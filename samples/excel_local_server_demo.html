다운로드와 Blob 처리

exportToXlsx

현재 피벗 보고서를 브라우저에서 엑셀 파일로 다운로드합니다. 파일을 만들기 위한 서버 처리는 필요하지 않습니다.

localControl: true를 설정하면 자동 다운로드 대신 localControlFunc 콜백으로 Blob을 전달합니다. 받은 Blob은 파일로 저장하거나 서버로 전송하는 등 원하는 방식으로 처리할 수 있습니다.

파일 저장 준비

브라우저에서 파일을 저장하려면 FileSaver.js를 불러옵니다. MIT 라이선스로 제공되며, 정품과 평가판 패키지의 export_server_samples 폴더에 포함되어 있습니다. 파일 위치에 맞게 경로를 지정하세요. 

<!-- FileSaver.js를 먼저 불러옵니다. -->
<script src="../export_server_samples/FileSaver.min.js"></script>

엑셀 파일 바로 다운로드

// 현재 피벗 보고서를 엑셀 파일로 바로 다운로드합니다.
function exportToLocal() {
    AUIPivot.exportToXlsx(myPivotID);
}

Blob을 받아 직접 처리

이 예제는 받은 Blob을 다운로드하고, Blob과 Base64 문자열을 개발자 도구의 콘솔에 출력합니다.

// 엑셀 Blob을 콜백으로 받아 다운로드하고 Base64 문자열도 확인합니다.
function getXlsxBlob() {
    AUIPivot.exportToXlsx(myPivotID, {
        // 자동 다운로드 대신 생성된 Blob을 아래 함수에 전달합니다.
        localControl: true,
        localControlFunc: function (blob) {
            console.log(blob);

            // FileSaver.js로 받은 Blob을 파일로 저장합니다.
            window.saveAs(blob, "얻어온 Blob 데이터 다운로딩.xlsx");

            // 파일 읽기가 끝나면 Data URL의 앞부분을 제외한 Base64를 출력합니다.
            const reader = new FileReader();
            reader.onload = function () {
                const base64data = reader.result.split(",")[1];
                console.log(base64data);
            };
            reader.readAsDataURL(blob);
        }
    });
}
