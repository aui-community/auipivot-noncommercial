Excel 파일명과 시트 이름

exportToXlsx

엑셀 내보내기 할 때 파일 이름(fileName)과 시트 이름(sheetName) 을 다음 처럼 지정 하십시오.

//엑셀로 저장
function exportToXlsx() {
	// 엑셀 내보내기 속성
	const exportProps = {
			// 파일 이름
			fileName : "2024년_분기별_지역_매출_보고서",
			// 엑셀 시트명
			sheetName : "2024년 분기별-지역별 매출 내역"
	};

	// 내보내기 실행
	AUIPivot.exportToXlsx(myPivotID, exportProps);
};
